    ![Multiset.replicate p.r 2 + {p.a}, Multiset.replicate p.s 3, Multiset.replicate p.t 3 + {p.b, p.c}]

def AK14.Transitive : {n : ℕ} → AK14.Triple n → Prop :=
fun {n} T => ∀ (x y : Fin n), ∃ g ∈ AK14.generated T, g x = y
def AK14.SameDessin : {n : ℕ} → AK14.Triple n → AK14.Triple n → Prop :=
fun {n} T U => ∃ q, ∀ (i : Fin 3), U i = q * T i * q⁻¹
def AK14.OrientedUniqueness : Prop :=
∀ (p : AK14.Passport) (n : ℕ),
  p.Admissible n → ∀ (T U : AK14.Triple n), AK14.Realizes p T → AK14.Realizes p U → AK14.SameDessin T U
def AK14.Theorem13Existence : Prop :=
∀ (p : AK14.Passport) (n : ℕ), p.Admissible n → ∃ T, AK14.Realizes p T
def AK14.Conjecture14 : Prop :=
∀ (p : AK14.Passport) (n : ℕ),
